var CMD_URL = 'https://raw.githubusercontent.com/bizminwoo/mdash-s9p6hu5g/main/ad-command.json';
var TEMPLATE = '자동광고템플릿';
var MAX_BUDGET_WON = 1000000; // 안전장치: 1회 총예산 최대 100만원 (초과 명령은 거부) — 2026-10-05 민우님 지시로 20만→100만
var FIXED_AG = '자동광고';   // 캠페인마다 재사용하는 광고그룹 이름 (입찰가 유지)

// 날짜 계산은 스크립트 서버 시간대와 무관하게 '달력 날짜'만 쓴다 (UTC 기준 산술)
function ymd(d) {
  var y = d.getUTCFullYear(), m = d.getUTCMonth() + 1, day = d.getUTCDate();
  return '' + y + (m < 10 ? '0' : '') + m + (day < 10 ? '0' : '') + day;
}
function calDate(str) { var p = str.split('-'); return new Date(Date.UTC(+p[0], +p[1] - 1, +p[2])); }
function kstDate(str) { return new Date(str + 'T00:00:00+09:00'); }
function todayKst() { var t = new Date(Date.now() + 9 * 3600000); return ymd(t); }
// 캠페인 시작일 ~ 명령 시작 전날까지 쓴 돈 (총예산 계산용)
function priorSpend(camp, startStr) {
  var sd = camp.getStartDate();
  if (!sd) return 0;
  var from = '' + sd.year + (sd.month < 10 ? '0' : '') + sd.month + (sd.day < 10 ? '0' : '') + sd.day;
  var to = ymd(new Date(calDate(startStr).getTime() - 86400000));
  if (to < from) return 0;
  return camp.getStatsFor(from, to).getCost();
}
// 구글 광고 문구 길이 제한: 한글 등 전각 문자는 2칸으로 계산 → 폭 기준으로 자른다
function cutWidth(str, maxWidth) {
  var out = '', w = 0;
  for (var i = 0; i < str.length; i++) {
    var c = str.charAt(i), cw = c.charCodeAt(0) > 255 ? 2 : 1;
    if (w + cw > maxWidth) break;
    out += c; w += cw;
  }
  while (out.length && ' -([,'.indexOf(out.charAt(out.length - 1)) >= 0) { out = out.substring(0, out.length - 1); }
  return out;
}

function main() {
  var raw = null;
  try {
    raw = JSON.parse(UrlFetchApp.fetch(CMD_URL, { muteHttpExceptions: true }).getContentText());
  } catch (e) {
    Logger.log('명령 파일 읽기 실패: ' + e);
    return;
  }
  var cmds = Array.isArray(raw) ? raw : [raw];
  for (var i = 0; i < cmds.length; i++) {
    try { runOne(cmds[i]); } catch (e) { Logger.log('명령 처리 오류(' + (cmds[i] && cmds[i].id) + '): ' + e); }
  }
}

function runOne(cmd) {
  var tmpl = (cmd && cmd.template) || TEMPLATE;
  var it = AdsApp.videoCampaigns().withCondition("campaign.name = '" + tmpl + "'").get();
  if (!it.hasNext()) { Logger.log('템플릿 캠페인(' + tmpl + ') 없음 — 대기'); return; }
  var camp = it.next();

  if (!cmd || !cmd.id || !cmd.videoId || !cmd.days || !cmd.totalBudgetWon || !cmd.startDate) {
    Logger.log('유효한 명령 없음');
    manageEnd(camp, cmd);
    return;
  }

  var marker = 'cmd-' + cmd.id;
  // v2.17: 전 세계 노출 (민우님 기본 설정: 지역 제한 없음 + 타겟 CPV 1원)
  if (cmd.allCountries) {
    var locIt = camp.targeting().targetedLocations().get(), nLoc = 0;
    while (locIt.hasNext()) { locIt.next().remove(); nLoc++; }
    if (nLoc) Logger.log('지역 설정 ' + nLoc + '개 삭제 → 전 세계 노출: ' + tmpl);
  }
  if (cmd.totalBudgetWon > MAX_BUDGET_WON) {
    Logger.log('안전장치: 예산 ' + cmd.totalBudgetWon + '원이 한도(' + MAX_BUDGET_WON + ')를 초과 — 실행 거부');
    return;
  }
  // 종료일 = 시작일 + (일수 - 1)  (구글 종료일은 "마지막으로 도는 날" 포함)
  var endDate = new Date(calDate(cmd.startDate).getTime() + (cmd.days - 1) * 86400000);
  if (ymd(endDate) < todayKst()) {
    Logger.log('종료일(' + ymd(endDate) + ')이 이미 지난 명령 — 실행 안 함');
    manageEnd(camp, cmd);
    return;
  }
  if (todayKst() < ymd(calDate(cmd.startDate))) {
    Logger.log('시작일(' + cmd.startDate + ') 전 — 대기: ' + marker);
    return;
  }

  // 1) 광고그룹: 새로 만들지 않고 있던 그룹 재사용, 이름도 안 바꿈 (v2.20 — 10/4까지 잘 되던 방식)
  var want_ag = cutWidth(cmd.title || marker, 250);
  var agsIt = camp.videoAdGroups().get(), all = [], agByVideo = null, agByCmd = null, agByTitle = null, agFixed = null, agOn = null, agUi = null;
  while (agsIt.hasNext()) {
    var g = agsIt.next(); all.push(g); var nm = g.getName();
    if (!agByVideo) { var vIt = g.videoAds().get(); while (vIt.hasNext()) { if (vIt.next().getVideoId() === cmd.videoId) { agByVideo = g; break; } } }
    if (cmd.adGroup && nm === cmd.adGroup && !agByCmd) agByCmd = g;
    if (nm === want_ag && !agByTitle) agByTitle = g;
    if (nm === FIXED_AG && !agFixed) agFixed = g;
    if (g.isEnabled() && !agOn) agOn = g;
    if (nm.indexOf('cmd-') !== 0 && !agUi) agUi = g;
  }
  var ag = agByVideo || agByCmd || agByTitle || agFixed || agOn || agUi || all[0] || null;
  if (!ag) {
    var agOp = camp.newVideoAdGroupBuilder().withName(want_ag).withAdGroupType('VIDEO_RESPONSIVE').build();
    if (!agOp.isSuccessful()) { Logger.log('광고그룹 생성 실패: ' + agOp.getErrors().join(', ')); return; }
    ag = agOp.getResult();
    Logger.log('🆕 캠페인에 광고그룹이 하나도 없어 새로 생성: ' + tmpl + ' / ' + want_ag);
  }
  want_ag = ag.getName();
  Logger.log('광고그룹 사용: ' + tmpl + ' / ' + want_ag);
  // v2.19: 이 명령으로 이미 한 번 켠 적이 있으면(라벨) 켜고 끄기는 건드리지 않는다 — 민우님이 화면에서 끈 걸 다시 켜지 않게
  var startedLbl = cutWidth('auto-started-' + cmd.id, 80);
  var started = hasLabel(ag, startedLbl);
  if (started) {
    Logger.log('이미 시작한 명령 — 켜기/끄기는 화면 설정 그대로 둠: ' + marker + (ag.isEnabled() ? '' : ' (광고그룹 꺼져 있음)'));
  } else {
    var others = all.filter(function (x) { return x.getName() !== want_ag; });
    for (var k = 0; k < others.length; k++) { if (others[k].isEnabled()) { others[k].pause(); Logger.log('다른 곡 광고그룹 일시중지(삭제 안 함): ' + others[k].getName()); } }
    if (!ag.isEnabled()) { ag.enable(); }
  }

  // 2) 이 곡 광고가 이미 켜져 있으면 그대로 둔다 (영상으로 판별 — 화면에서 문구를 고쳐도 같은 광고로 인식)
  var mine = null, adsIt = ag.videoAds().get(), adList = [];
  while (adsIt.hasNext()) { var a = adsIt.next(); adList.push(a); if (a.getVideoId() === cmd.videoId && (!mine || a.isEnabled())) mine = a; }
  var isNew = !(mine && mine.isEnabled());
  if (isNew) {
    // 새 명령 시작: 종료일을 맞추고, 다른 곡 광고는 끈다
    camp.setEndDate(ymd(endDate));
    for (var j = 0; j < adList.length; j++) { if (adList[j] !== mine && adList[j].isEnabled()) { adList[j].pause(); Logger.log('같은 그룹의 다른 광고 일시중지(삭제 안 함)'); } }
    if (mine) { mine.enable(); Logger.log('기존 광고 다시 켬: ' + marker); }
  }
  if (!isNew) { for (var r = 0; r < adList.length; r++) { if (adList[r] !== mine && adList[r].isEnabled()) { adList[r].pause(); Logger.log('남은 다른 광고 일시중지(삭제 안 함)'); } } }
  // 총예산은 "캠페인 시작일부터 전체" 기준이라, 이 명령 전까지 이미 쓴 돈 + 이번 예산으로 맞춘다 (v2.10)
  // (2026-09-28 사고: 9/14 시작 캠페인에 5만원만 걸어 이미 쓴 10만원 때문에 Fishing Ground가 하루 종일 0회 노출)
  var want = Math.round(priorSpend(camp, cmd.startDate) + cmd.totalBudgetWon);
  var budget = camp.getBudget(), cur = null;
  try { cur = budget.getTotalAmount(); } catch (e) {}
  if (cur == null || Math.abs(cur - want) >= 1) {
    budget.setTotalAmount(want);
    Logger.log('총예산 ' + want + '원으로 설정 (이전 사용분 ' + (want - cmd.totalBudgetWon) + '원 + 이번 ' + cmd.totalBudgetWon + '원): ' + marker);
  }
  if (!mine) {
    // 영상 자산: 이미 등록된 유튜브 영상 자산이 있으면 재사용, 없으면 새로 등록 (adMedia는 2026년 폐지됨)
    var videoAsset = null;
    var assetIt = AdsApp.adAssets().assets()
      .withCondition("asset.type = YOUTUBE_VIDEO")
      .withCondition("asset.youtube_video_asset.youtube_video_id = '" + cmd.videoId + "'")
      .get();
    if (assetIt.hasNext()) {
      videoAsset = assetIt.next();
    } else {
      var assetOp = AdsApp.adAssets().newYouTubeVideoAssetBuilder()
        .withName('auto-' + cmd.videoId)
        .withYouTubeVideoId(cmd.videoId)
        .build();
      if (!assetOp.isSuccessful()) { Logger.log('영상 등록 실패: ' + assetOp.getErrors().join(', ')); return; }
      videoAsset = assetOp.getResult();
      Logger.log('영상 자산 등록: ' + cmd.videoId);
      if (AdsApp.getExecutionInfo().isPreview()) {
        Logger.log('[미리보기] 영상 자산은 미리보기에서 실제 생성되지 않아(임시 ID) 광고 생성은 실제 실행 때 진행됩니다 — 여기까지 오류 없으면 정상');
        return;
      }
      // 실제 실행: 방금 만든 자산을 다시 조회해 확정된 자산으로 사용
      var reIt = AdsApp.adAssets().assets()
        .withCondition("asset.type = YOUTUBE_VIDEO")
        .withCondition("asset.youtube_video_asset.youtube_video_id = '" + cmd.videoId + "'")
        .get();
      if (reIt.hasNext()) { videoAsset = reIt.next(); }
    }
    // 로고 이미지(반응형 영상 광고 필수 항목): 계정에 있는 1:1 이미지 자산 'OOW SOUND'(800x800, id 348514433077) 사용
    var logoIt = AdsApp.adAssets().assets().withCondition("asset.id = " + (cmd.logoAssetId || 348514433077)).get();
    if (!logoIt.hasNext()) { Logger.log('로고 이미지 자산을 찾지 못함 — 광고 생성 중단'); return; }
    var logoAsset = logoIt.next();
    var title = cutWidth(cmd.title || '신곡 듣기', 90);          // 긴 광고 제목: 최대 90폭
    var artist = cutWidth((cmd.title || '신곡').split(' - ')[0], 15); // 짧은 제목: 최대 15폭 (한글 7자)
    var adName = cutWidth(cmd.title || marker, 250);
    // v2.17: 구글 API로 직접 생성 — 비즈니스 이름(필수)·로고(필수)·긴 제목·설명·영상·링크. 화면에서 만든 광고와 같은 구성(짧은 제목·버튼 없음).
    var cid = AdsApp.currentAccount().getCustomerId().replace(/-/g, '');
    var resOf = function (obj, kind) { try { var rn = obj.getResourceName && obj.getResourceName(); if (rn) return rn; } catch (e) {} return 'customers/' + cid + '/' + kind + '/' + obj.getId(); };
    var vra = {
      videos: [{ asset: resOf(videoAsset, 'assets') }],
      longHeadlines: [{ text: title }],
      descriptions: [{ text: cutWidth(cmd.title || '신곡', 70) }],
      businessName: { text: cutWidth(cmd.businessName || 'oowsound', 25) },
      logoImages: [{ asset: resOf(logoAsset, 'assets') }]
    };
    if (cmd.withButton) { vra.headlines = [{ text: artist }]; vra.callToActions = [{ text: '듣기' }]; }
    var mres = null, okMutate = false, mErr = '';
    // ⓪ 10/4까지 잘 되던 그대로: 빌더 + 짧은 제목 + 버튼 + 로고
    try {
      var oop = ag.newVideoAd().responsiveVideoAdBuilder().withAdName(adName).withVideo(videoAsset)
        .withFinalUrl('https://www.youtube.com/watch?v=' + cmd.videoId).withLongHeadline(title).withHeadline(artist)
        .withDescription(cutWidth(cmd.title || '신곡', 70)).withCallToAction('듣기').withLogoImages([logoAsset]).build();
      okMutate = oop.isSuccessful();
      if (okMutate) Logger.log('광고 생성(예전 방식): ' + adName); else mErr = '예전 방식: ' + oop.getErrors().join(', ');
    } catch (e) { mErr = '예전 방식: ' + e; }
    if (!okMutate && mErr) Logger.log('예전 방식 실패 → 다른 방식 시도: ' + mErr);
    // ① 빌더에 비즈니스 이름 기능이 생겼으면 빌더로 (구글이 스크립트를 고친 경우)
    var probe = ag.newVideoAd().responsiveVideoAdBuilder();
    if (!okMutate && typeof probe.withBusinessName === 'function') {
      var bop = probe.withAdName(adName).withVideo(videoAsset).withFinalUrl('https://www.youtube.com/watch?v=' + cmd.videoId)
        .withLongHeadline(title).withDescription(cutWidth(cmd.title || '신곡', 70)).withCallToAction('듣기')
        .withBusinessName(cutWidth(cmd.businessName || 'oowsound', 25)).withLogoImages([logoAsset]).build();
      okMutate = bop.isSuccessful(); if (!okMutate) mErr = '빌더: ' + bop.getErrors().join(', ');
    }
    if (!okMutate) try {
      mres = AdsApp.mutate({ adGroupAdOperation: { create: {
        adGroup: resOf(ag, 'adGroups'), status: 'ENABLED',
        ad: { name: adName, finalUrls: ['https://www.youtube.com/watch?v=' + cmd.videoId], videoResponsiveAd: vra } } } });
      okMutate = mres.isSuccessful();
      if (!okMutate) mErr = (mres.getErrorMessages ? mres.getErrorMessages().join(', ') : '알 수 없음');
    } catch (e) { mErr = '' + e; }
    if (!okMutate) {
      Logger.log('광고 생성 실패(API 직접): ' + mErr + ' / 영상자산 ' + videoAsset.getId() + ' / 로고 ' + logoAsset.getId() + ' / 그룹 ' + ag.getName());
      // ③ 예산·기간은 맞춰뒀으니 광고만 화면에서: 이 캠페인 아무 그룹에 이 영상 광고를 만들면 다음 실행 때 자동으로 켬
      if (camp.isEnabled() && !mine) { /* 다른 곡 광고가 도는 중일 수 있으니 캠페인 상태는 건드리지 않음 */ }
      Logger.log('👉 광고만 화면에서 만들어주세요: 「' + tmpl + '」 캠페인에 영상 ' + cmd.videoId + ' 광고(비즈니스 이름 oowsound·로고) → 다음 정시 실행 때 자동으로 켜짐');
      return;
    }
    Logger.log('광고 생성: ' + marker + ' / 영상 ' + cmd.videoId);
  }

  // 3) 종료일·총예산 재확인 후 캠페인 켜기
  if (camp.getEndDate() == null || ymd(new Date(Date.UTC(camp.getEndDate().year, camp.getEndDate().month - 1, camp.getEndDate().day))) != ymd(endDate)) {
    camp.setEndDate(ymd(endDate));
  }
  if (cmd.go === false) {
    // v2.11: 확인 대기 — 준비만 하고 켜지 않음 (민우님이 "ㄱ" 하면 go:true로 바뀜)
    if (camp.isEnabled()) camp.pause();
    Logger.log('⏸ 준비 완료, 켜기 대기(go:false): ' + marker);
    return;
  }
  if (started) {
    if (!camp.isEnabled()) Logger.log('캠페인 꺼져 있음(화면에서 끈 것으로 보고 그대로 둠): ' + marker);
    else Logger.log('이미 집행 중: ' + marker);
  } else if (!camp.isEnabled()) {
    camp.enable();
    Logger.log('✅ 광고 시작: ' + marker + ' / 총예산 ' + cmd.totalBudgetWon + '원 / ' + cmd.startDate + '~' + ymd(endDate));
  } else {
    Logger.log('이미 집행 중: ' + marker);
  }
  if (!started && !AdsApp.getExecutionInfo().isPreview()) markStarted(ag, startedLbl);
  manageEnd(camp, cmd);
}

// v2.19: 명령별 '한 번 켰음' 표시 (광고그룹 라벨). 라벨 기능이 안 되면 예전처럼 동작(매번 켬)하고 로그만 남김.
function hasLabel(ent, name) {
  try { return ent.labels().withCondition("label.name = '" + name + "'").get().hasNext(); }
  catch (e) { Logger.log('라벨 확인 실패(예전 방식으로 진행): ' + e); return false; }
}
function markStarted(ent, name) {
  try {
    if (!AdsApp.labels().withCondition("label.name = '" + name + "'").get().hasNext()) AdsApp.createLabel(name);
    ent.applyLabel(name);
  } catch (e) { Logger.log('라벨 붙이기 실패(다음 실행에도 켜기 시도함): ' + e); }
}

function manageEnd(camp, cmd) {
  if (!cmd || !cmd.startDate || !cmd.days) { return; }
  var end = new Date(kstDate(cmd.startDate).getTime() + cmd.days * 86400000);
  if (new Date().getTime() >= end.getTime() && camp.isEnabled()) {
    camp.pause();
    Logger.log('⏹ 기간 종료 — 캠페인 일시중지');
  }
}
